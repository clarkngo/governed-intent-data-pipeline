# gidp.policy.v1 — evaluated on every compile
identity:
  analyst: [SELECT, AGGREGATE]
  admin:   [SELECT, AGGREGATE, APPROVE_HITL]
catalog:
  approved_for_analytics: [subscriptions, support_tickets]
  join_only_non_pii:      [customers.region, customers.customer_id, customers.status]
  restricted:             [customers.ssn]          # never overridable
  confidential_pii:       [customers.email, customers.name]
tools:
  allow: [duckdb.select, duckdb.aggregate, artifact.publish]
  deny:  [export.raw, copy_to, ddl, dml, external_http]
injection:
  block: [ignore_policies, bypass_guardrails, jailbreak]
quotas:
  max_rows_returned: 5000
  max_compile_per_session: 40
quality:
  require: [row_count > 0, no_pii_in_projection, retention_pct in 0..100]
